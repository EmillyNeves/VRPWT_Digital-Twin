

\begin{algorithm}
\caption{ILS - Iterated Local Search}
\label{alg:ils}
\SetAlgoLined
\Entrada{O grafo $G = (V, E)$, $Construtivo = \{AC_{\text{ILS}}, RBEP\}$, $OrdenacaoBusca = \{Grau, PageRank\}$, $ModoPertubacao = \{Grau, Pagerank\}$ e $SelecaoVizinhos = \{Primeiro,  Melhor\}$.} \label{lin:EntradaILS}
\Saida{Uma solução válida $T$.}
    $O_D \gets Decomposicao\_por\_Pontes(G)$; \label{lin:DecomposicaoPorPonteILS} \\
    $T_0 \gets Solucao\_Inicial(G, O_D, Construtivo)$; \label{lin:SolucaoInicialILS} \\ 
    $T^* \gets Busca\_Local(T_0, O_D, OrdenacaoBusca, SelecaoVizinhos)$; \label{lin:BuscaLocal1ILS}\\
    \Repita{$condicao\ de\ parada\ atendida$}{\label{lin:InicioLoopILS}
        $T' \gets Perturbacao(T^*, O_D, ModoPertubacao)$; \label{lin:PertubacaoILS}\\
        $T'' \gets Busca\_Local(T', O_D, OrdenacaoBusca, SelecaoVizinhos)$; \label{lin:BuscaLocal2ILS}\\
        $T^* \gets Criterio\_Aceitacao(T^*, T'', historia)$; \label{lin:CriterioAceitacaoILS}\\   
    }\label{lin:FimLoopILS}
\textbf{return} $T^*$\;
\end{algorithm}
