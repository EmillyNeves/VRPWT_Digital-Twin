\begin{algorithm}[H]
\caption{Custom Neighbor}
\label{alg:customNeighbor}
\SetAlgoLined
\Entrada {Um grafo $G = (V, E)$ sem pontes, a solução atual $T_{\text{curr}}$, o conjunto de vértices obrigatórios $O_{\text{D}}$, $OrdenacaoBusca = \{Grau, PageRank\}$ e $SelecaoVizinhos = \{Primeiro,  Melhor\}$.}  
\Saida {A solução $T$.}
    $T \gets T_{\text{curr}}$\;
    $T_{\text{best}} \gets T_{\text{curr}}$\;
    $L_{\text{D}} \gets O_{\text{D}}$\;
    $DB \gets Sort\_Ascending(G, T_{\text{curr}}, O_{\text{D}}, OrdenacaoBusca)$\;   
   \Para {$v \in DB$}{
        \Para {$u \in N_{\text{T}}(v)$}{
            Remove a aresta $\{u, v\}$ de $T$\;
            $e \gets \text{Find\_Edge}(u, v, T, L_{\text{D}})$\;
            \eSe{($e \ne 0$)}{
                Insere a aresta $e$ em $T$\;
            }{
                Retorna a aresta $\{u, v\}$ para $T$\;
            }

            \Se{(($d_T(v) + l(v) \leq d) \textbf{ ou } \left((d_T(u) + l(u) \leq d) \land (u \in DB)\right)$)}{
                \uSe{$SelecaoVizinhos = \text{Primeiro}$}{
                    \textbf{return} $T$\;
                }
                \SenaoSe{$DB_T < DB_T_{\text{best}}$}{
                        $T_{\text{best}} \gets T$\;             
                }
            }
        }
        $L_{\text{D}} \gets L_{\text{D}} \cup \{v\}$\;
   }
\Retorna{$T_{\text{best}}$.}
\end{algorithm}
