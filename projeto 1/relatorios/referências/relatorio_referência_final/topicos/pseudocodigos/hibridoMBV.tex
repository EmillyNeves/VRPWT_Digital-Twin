\begin{algorithm}
\caption{Híbrido MBV}
\label{alg:frameworkMBV}
\SetAlgoLined
\Entrada{O grafo $G = (V, E)$, $\alpha = (0, 1)$, $MaxIter$.} \label{lin:EntradaILS}
\Saida{Uma solução válida $T$.}
\Para{$i = 1$ \textbf{até} $MaxIter$}{ 
    $T_0 \gets RPREV(G, \alpha)$; \label{lin:SolucaoInicialILS} \\ 
    $T_{local} \gets Busca\_Local(T_0)$; \label{lin:BuscaLocal1ILS}\\
    \Repita{$condicao\ de\ parada\ atendida$}{\label{lin:InicioLoopILS}
        $T' \gets Perturbacao(T_{local})$; \label{lin:PertubacaoILS}\\
        $T'' \gets Busca\_Local(T')$; \label{lin:BuscaLocal2ILS}\\
        $T_{local} \gets Criterio\_Aceitacao(T^*, T'', historia)$; \label{lin:CriterioAceitacaoILS}\\   
    }\label{lin:FimLoopILS}
    \Se{$T_{local}$ é melhor que $T^*$}{ \label{lin:MultiStart_IfBetter}
        $T^* \gets T_{local}$; \label{lin:MultiStart_UpdateTstar}\\
    }
}
\textbf{return} $T^*$\;
\end{algorithm}
